Product IO

Read Product

  from snapkit import Product
  product = Product.read('/path/to/product.dim')
  # or
  from snapkit import ProductIO
  product = ProductIO.read('/path/to/product.tif')

Write Product

  from snapkit import Product
  product = Product('My Product', 'MyType', [10,10])
  product.saveAs('/path/to/output.tif', 'GeoTIFF')
  # or
  from snapkit import ProductIO
  ProductIO.save(product, product.file_location)

Data Format

  from snapkit import ProductIO
  in_formats = ProductIO.input_formats()
  print(in_formats)
  out_formats = ProductIO.output_formats()
  print(out_formats)

Graph Processing

Setup and Execute a Graph

  from snapkit import Graph

  # Create a graph
  graph = Graph()

  # Add operators
  read_id = graph.read("path/S1/file.zip")
  calibrate_id = graph.add_operator("calibrate", "Calibration", {
      "outputSigmaBand": True,
      "outputBetaBand": False,
      "outputGammaBand": False
  })
  terrain_id = graph.add_operator("terrain", "Terrain-Correction", {
      "demName": "SRTM 3Sec",
      "pixelSpacingInMeter": 10.0
  })
  write_id = graph.write("output.dim")

  # Connect operators
  graph.connect(read_id, calibrate_id, terrain_id, write_id)

  print("Graph has been setup and connected.")

  # export the graph as XML file
  graph.export_xml("myGraph.xml")

  product = graph.execute()
  product.saveAs("output.dim", "NetCDF4-CF")

Band Computation

Compute Band

  from snapkit import SNAP, Raster
  from math import nan

  b8 = product["B8"].fetch(geophysical=True)
  b4 = product["B4"].fetch(geophysical=True)

  ndvi = product.add_band("ndvi", Raster.TYPE_FLOAT32)

  ndvi_data = ndvi.fetch()
  for i, (x, y) in enumerate(zip(b8, b4)):
      d = x + y
      ndvi_data[i] = (x - y) / d if d != 0 else nan

  ndvi.apply(ndvi_data)

Compute Band with numpy

  from snapkit import SNAP, Raster
  import numpy as np

  b8 = product["B8"].fetch(geophysical=True)
  b4 = product["B4"].fetch(geophysical=True)

  b8 = np.asarray(b8, dtype=float)
  b4 = np.asarray(b4, dtype=float)

  den = b8 + b4
  with np.errstate(divide='ignore', invalid='ignore'):
      ndvi_data = (b8 - b4) / den
      ndvi_data[den == 0] = np.nan

  ndvi = product.add_band("ndvi", Raster.float64)
  ndvi.apply(ndvi_data)

SNAP Interaction

  from snapkit import SNAP
  # check if SNAP is available, if script is running from command line SNAP is not available.
  if SNAP.is_available():
    # get the currently selected product
    product = SNAP.selected_product()

    # open a raster in SNAP
    SNAP.view_raster("Sigma0_VH", product)

    # open a RGB composite in SNAP
    SNAP.view_rgb('B4', 'B3', 'B2', product, 'view_name')
